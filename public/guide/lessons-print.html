บทเรียนเพิ่มเติม: การเขียนโปรแกรมและความรู้ที่ใช้ในโจทย์

บทเรียนนี้ต่อจากเนื้อหาสามกลุ่มเดิม เริ่มอ่านตามลำดับแล้วทดลองพิมพ์โปรแกรมด้วยตนเอง ข้อมูลเข้าและผลลัพธ์ในตัวอย่างเป็นเพียงตัวอย่าง ต้องทำตามรูปแบบของโจทย์จริงเสมอ

1. จากปัญหาสู่โปรแกรม

2. โปรแกรมแรกและวิธีรัน

Python ทำงานผ่านตัวแปล ใช้การเยื้องบรรทัดกำหนดขอบเขต ส่วน C แปลเป็นโปรแกรมเครื่องก่อน และ Java แปลเป็น bytecode เพื่อให้ JVM ทำงาน แต่ละระบบต้องติดตั้งเครื่องมือที่เหมาะสมก่อน คำสั่งตัวอย่างต่อไปนี้รันใน terminal จากโฟลเดอร์ที่บันทึกไฟล์

Python: hello.py

print("Hello, ICT!")
# รัน: python hello.py

C: hello.c

#include <stdio.h>
int main(void) {
    printf("Hello, ICT!\n");
    return 0;
}
/* คอมไพล์: gcc hello.c -o hello
   รัน: ./hello   (Windows อาจใช้ .\hello.exe) */

Java: Main.java

public class Main {
    public static void main(String[] args) {
        System.out.println("Hello, ICT!");
    }
}
// คอมไพล์: javac Main.java
// รัน: java Main

ใน Java ชื่อไฟล์ของ public class ต้องตรงกับชื่อคลาส รวมตัวพิมพ์ใหญ่เล็ก ใน C และ Java เครื่องหมาย ; ปิดคำสั่ง ส่วน { } ครอบบล็อก โค้ดตัวอย่างสำหรับแต่ละภาษาเป็นโปรแกรมแยกกัน

3. ตัวแปร ชนิดข้อมูล และรับข้อมูล

ตัวแปรเก็บค่า ชนิดข้อมูลบอกว่าจัดเก็บและใช้ค่าชนิดใด เช่น จำนวนเต็ม int ทศนิยม double/float ข้อความ string และค่าจริงเท็จ boolean ชื่อมีความหมาย ห้ามเริ่มด้วยตัวเลข และตัวพิมพ์ใหญ่เล็กต่างกัน ตรวจว่ารับข้อความแล้วแปลงเป็นตัวเลขก่อนคำนวณ

Python

name = input().strip()
age = int(input())
score = float(input())
print(name, age + 1, score)

C

#include <stdio.h>
int main(void) {
    int age;
    double score;
    if (scanf("%d %lf", &age, &score) != 2) return 1;
    printf("%d %.2f\n", age + 1, score);
    return 0;
}

ใน C scanf ต้องส่งตำแหน่งตัวแปรด้วย & สำหรับตัวเลข และ %lf ใช้รับ double ส่วน printf ใช้ %.2f แสดงทศนิยมสองตำแหน่ง ตรวจผลรับข้อมูลเพื่อไม่ใช้ค่าที่ไม่ได้กำหนด

Java

import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int age = in.nextInt();
        double score = in.nextDouble();
        System.out.printf("%d %.2f%n", age + 1, score);
    }
}

Python int(input()) อ่านหนึ่งบรรทัด หากโจทย์ให้หลายตัวเลขในบรรทัดเดียวใช้ a, b = map(int, input().split()) Java ใช้ Scanner.nextInt() อ่านตัวเลขทีละตัว C ใช้ scanf ได้หลายค่า อย่าพิมพ์คำชวนป้อนข้อมูลถ้าระบบตรวจอัตโนมัติไม่ได้ขอ

Python ใช้ // หารปัดลง ขณะที่ C และ Java หารจำนวนเต็มตัดส่วนทศนิยมเข้าหาศูนย์; ค่าลบจึงอาจได้ต่างกัน หากต้องการทศนิยมใช้ชนิดทศนิยมอย่างน้อยหนึ่งตัว เช่น 5.0 / 2

5. การทำซ้ำและการไล่ค่า

for เหมาะเมื่อทราบช่วง ส่วน while ทำซ้ำตราบใดที่เงื่อนไขเป็นจริง ต้องมีการเปลี่ยนค่าที่นำไปสู่การหยุด ระวังขอบเขต: range(1, 6) ให้ 1–5 ส่วน range(5) ให้ 0–4

# Python: รวมเลขคู่ 1 ถึง n
n = int(input())
total = 0
for i in range(1, n + 1):
    if i % 2 == 0:
        total += i
print(total)
/* C: หลังรับ int n แล้ว */
int total = 0;
for (int i = 1; i <= n; i++) {
    if (i % 2 == 0) total += i;
}
printf("%d\n", total);
// Java: หลังรับ int n แล้ว
int total = 0;
for (int i = 1; i <= n; i++) {
    if (i % 2 == 0) total += i;
}
System.out.println(total);

เมื่อ n=6 ผลคือ 2+4+6=12 break ออกจากลูปปัจจุบัน ส่วน continue ข้ามไปเริ่มรอบถัดไป ลูปซ้อนต้องติดตามว่าลูปด้านในเริ่มใหม่กี่ครั้ง หากสองลูปวิ่ง n รอบเต็มจะทำงานประมาณ n² ครั้ง

6. อาร์เรย์ รายการ ข้อความ และการเปลี่ยนค่า

Index เริ่มที่ 0: ตัวแรกตำแหน่ง 0 ตัวสุดท้ายตำแหน่ง n−1 การเข้าถึงนอกช่วงเป็นข้อผิดพลาด Python list ปรับขนาดได้ C array ขนาดคงที่ตามที่ประกาศ Java array มีความยาวคงที่ ส่วน ArrayList เพิ่มสมาชิกได้

# Python
values = [5, 11, 2, 8, 15]
values.append(7)
for i in range(len(values)):
    print(i, values[i])
/* C */
int values[] = {5, 11, 2, 8, 15};
int n = sizeof values / sizeof values[0];
for (int i = 0; i < n; i++) printf("%d\n", values[i]);
// Java
int[] values = {5, 11, 2, 8, 15};
for (int i = 0; i < values.length; i++) {
    System.out.println(values[i]);
}

การสลับสองค่าใน C/Java ใช้ตัวแปรชั่วคราว temp = a; a = b; b = temp; Python ใช้ a, b = b, a ได้ ใน Python การส่ง list ให้ฟังก์ชันแล้ว append เปลี่ยน list เดิม แต่การกำหนดตัวแปรจำนวนเต็มใหม่ภายในฟังก์ชันไม่ได้เปลี่ยนตัวแปรจำนวนเต็มของผู้เรียก ต้อง return ผลกลับไป

ข้อความ Python ใช้ len(s), s[i], s.lower(); Java ใช้ s.length(), s.charAt(i), s.equals(other) เพื่อเทียบเนื้อหา; C ใช้อาร์เรย์ char ลงท้ายด้วย '\0' และฟังก์ชันจาก <string.h> เช่น strlen และ strcmp ระวังพื้นที่และการสิ้นสุดข้อความ

7. ฟังก์ชันและ recursion

ฟังก์ชันแบ่งงาน ตั้งชื่อ รับพารามิเตอร์และอาจคืนค่า return ตัวแปรภายในมีขอบเขตของตัวเอง Recursion คือฟังก์ชันเรียกตัวเอง ต้องมี base case และแต่ละขั้นเข้าใกล้ฐาน ไม่เช่นนั้น stack อาจเต็ม

# Python
def factorial(n):
    if n <= 1:
        return 1
    return n * factorial(n - 1)
print(factorial(4))  # 24
/* C: วางฟังก์ชันก่อน main */
int factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}
// Java: วางภายใน class Main
static int factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}

ตัวอย่างนี้สมมติ n เป็นจำนวนเต็มไม่ติดลบและผลไม่เกินช่วงชนิดข้อมูล หาก f(a,b) คืน 1 เมื่อ b≤0 นอกนั้นคืน a×f(a,b−2) แล้ว f(2,9)=2⁵=32 เพราะ b เป็น 9,7,5,3,1 ก่อนถึง −1

8. โครงสร้างข้อมูล OOP และการแก้บั๊ก

Stack เข้าหลังออกก่อน ใช้กับการเรียกฟังก์ชันและ undo; Queue เข้าก่อนออกก่อน ใช้กับคิวงาน; Dictionary/Map จับคู่ key กับ value; Set เก็บสมาชิกไม่ซ้ำ Linked list เชื่อมโหนดด้วยตัวอ้างอิง ส่วน tree เป็นโครงสร้างลำดับชั้น เลือกจากงานที่ต้องทำ ไม่ใช่จำแต่ชื่อ

OOP: class เป็นแม่แบบ object เป็นอินสแตนซ์ encapsulation รวมข้อมูลและพฤติกรรมพร้อมควบคุมการเข้าถึง inheritance สืบทอดความสามารถ polymorphism ใช้อินเทอร์เฟซเดียวกับพฤติกรรมต่างกัน abstraction มองสิ่งจำเป็นโดยซ่อนรายละเอียด โดย C เป็นภาษาเชิงกระบวนการ ส่วน Java และ Python รองรับคลาส

# Python
class Counter:
    def __init__(self):
        self.value = 0
    def increment(self):
        self.value += 1
c = Counter()
c.increment()
print(c.value)  # 1

Syntax error คือรูปแบบภาษาไม่ถูก Runtime error เกิดระหว่างรัน เช่น หารศูนย์หรือ index เกิน Logic error โปรแกรมรันได้แต่ผลผิด เริ่มตรวจจากข้อความ error หมายเลขบรรทัด ข้อมูลตัวอย่าง และพิมพ์ค่ากลางหรือใช้ debugger ทดลองกรณีว่าง ค่าต่ำสุด/สูงสุด และค่าซ้ำ

10. Library, import, package และ plugin

Library คือโค้ดที่นำกลับมาใช้ เช่น คณิตศาสตร์หรือจัดการข้อมูล Module คือหน่วยไฟล์หรือหน่วยโค้ด Package จัดกลุ่ม module ส่วน plugin คือส่วนเสริมที่แอปหรือเครื่องมือหลักเรียกใช้งานผ่านจุดต่อที่รองรับ ตัวอย่างส่วนขยาย editor ช่วยตรวจ syntax แต่ไม่ใช่สิ่งที่ต้อง import ในโปรแกรมเสมอไป

# Python: math เป็น standard library
import math
print(math.sqrt(25))
from collections import Counter
print(Counter(["a", "b", "a"])["a"])  # 2

แพ็กเกจภายนอกอาจติดตั้งด้วย python -m pip install package_name ภายใน virtual environment เช่น python -m venv .venv จากนั้นเปิดใช้งานตามระบบปฏิบัติการ ต้องใช้ชื่อแพ็กเกจที่เชื่อถือได้และตรวจเวอร์ชัน ในการแข่งขันให้ตรวจว่าระบบอนุญาต package ภายนอกหรือไม่ ห้ามสมมติว่ามีอินเทอร์เน็ตหรือติดตั้งเพิ่มได้

/* C: header ประกาศ interface */
#include <stdio.h>
#include <math.h>
/* บางระบบต้อง link math: gcc main.c -o main -lm */
// Java: import class จาก standard library
import java.util.Arrays;
import java.util.Scanner;

C #include ให้ preprocessor นำประกาศจาก header มาใช้ และการ link เชื่อม implementation ที่ต้องการ Java import ทำให้เรียกชื่อคลาสโดยไม่ต้องเขียนชื่อ package เต็มทุกครั้ง ไม่ได้ติดตั้ง library อัตโนมัติ Python import ค้นหา module จาก environment ของ interpreter หากพบ ModuleNotFoundError ให้ตรวจชื่อและ environment ก่อน

11. Data และ AI: ศัพท์เพิ่มเติมจากโจทย์

Embedding แทนคำหรือข้อมูลด้วยเวกเตอร์ Cosine similarity วัดมุมระหว่างเวกเตอร์ x,y ด้วย (x·y)/(|x||y|) โดยเวกเตอร์ต้องไม่เป็นศูนย์ ค่าใกล้ 1 แสดงทิศทางใกล้กัน ค่า 0 ตั้งฉาก และ −1 ตรงข้าม ความใกล้เชิงความหมายขึ้นกับโมเดลและบริบท ไม่ได้รับประกันจากสูตรเพียงอย่างเดียว

Encoder แปลงข้อมูลนำเข้าเป็นตัวแทนภายใน ส่วน decoder ใช้ตัวแทนหรือบริบทเพื่อสร้างผลลัพธ์ เช่น แปลข้อความ ทั้งนี้สถาปัตยกรรมแต่ละระบบต่างกัน ตัวแทนที่ส่งระหว่างส่วนอาจเป็นชุดเวกเตอร์ ไม่จำเป็นต้องเป็นเวกเตอร์เดียว One-hot encoding ของข้อมูลตารางเป็นอีกความหมายของคำว่า encoding

Softmax แปลงคะแนน z ของหลายคลาสเป็น pᵢ=e^zᵢ/Σe^zⱼ ผลแต่ละค่าอยู่ระหว่าง 0 ถึง 1 และรวมเป็น 1 ใช้บ่อยกับการจำแนกหนึ่งคลาสจากหลายคลาส เช่น logits [0,0] ให้ [0.5,0.5] แต่คะแนนที่ดูเป็นความน่าจะเป็นอาจยังไม่ calibrated

Expert system ใช้ knowledge base เก็บข้อเท็จจริงและกฎ ส่วน inference engine ใช้กฎอนุมานผล ตัวอย่าง IF ไข้สูง AND ไอ THEN แนะนำให้ประเมินเพิ่มเติม ระบบแบบกำหนดกฎไม่จำเป็นต้องเรียนรู้จากข้อมูล k-NN เป็น lazy learning เพราะมักเก็บตัวอย่างฝึกไว้ แล้วคำนวณระยะและเลือกเพื่อนบ้านตอนทำนาย จึงต้องคิดเรื่องสเกลและต้นทุนการทำนาย

12. ระบบ เครือข่าย และตรรกะเพิ่มเติม

Paging แบ่งหน่วยความจำเป็นหน้าขนาดคงที่ ส่วน segmentation แบ่งตามส่วนเชิงตรรกะที่ขนาดอาจต่างกัน Virtual memory ช่วยให้โปรแกรมใช้พื้นที่ที่อยู่เสมือนโดยระบบจัดการการแมป ระบบ time-sharing แบ่งเวลา CPU เป็นช่วงสั้นให้หลายงานสลับกัน GPU มีหน่วยประมวลผลขนานจำนวนมากเหมาะกับกราฟิกและงานเมทริกซ์ ส่วน simultaneous multithreading ให้หนึ่งคอร์จัดการหลายเธรดเชิงตรรกะ แต่ไม่ได้รับประกันความเร็วเพิ่มเป็นสองเท่า

Virtual machine จำลองเครื่องและมี guest OS ของตนบน hypervisor ส่วน container แยกโปรเซสและใช้ kernel ร่วมกับ host ที่เข้ากันได้ จึงมี overhead ต่างกัน แต่ขอบเขตความปลอดภัยและการใช้งานขึ้นกับระบบ Edge computing ประมวลผลใกล้แหล่งข้อมูลเพื่อลดเวลาหน่วงหรือการส่งข้อมูล ส่วน cloud ให้ทรัพยากรผ่านเครือข่าย การเพิ่มลดทรัพยากรตามความต้องการเรียก elasticity; load balancing กระจายงาน; replication เก็บสำเนาหลายตำแหน่ง ช่วยความพร้อมใช้งานและอาจเพิ่มความเร็วอ่าน แต่ไม่ใช่ backup เสมอไป

การแบ่ง IPv4 แบบ classful เดิมดู octet แรก: Class A 1–126, B 128–191, C 192–223 (127 ใช้ loopback); ปัจจุบันใช้ CIDR/subnet mask ในการกำหนดเครือข่ายจริง ไม่ควรใช้ class เดา mask ของเครือข่ายปัจจุบัน OSI ชั้น Data Link สร้าง frame และ switch ทำงานในชั้นนี้; HTTP อยู่ชั้น Application

Caesar cipher เลื่อนตัวอักษรตามจำนวนตำแหน่งและวน Z กลับ A เช่น UTCC เลื่อน 2 เป็น W V E E หรือ WVEE เป็นวิธีสอนแนวคิด ไม่ปลอดภัยสำหรับปกป้องข้อมูลจริง คำสั่ง cd ..\Sci บน Windows ย้ายไปโฟลเดอร์ Sci ซึ่งเป็นโฟลเดอร์พี่น้องจากตำแหน่งปัจจุบัน

โจทย์ตรรกะต้องใช้ทุกเงื่อนไขพร้อมกัน ถ้า A→B และ A จริง สรุป B ได้ แต่ถ้า B จริงเพียงอย่างเดียวสรุป A ไม่ได้ หากเงื่อนไขบังคับให้ข้อเท็จจริงหนึ่งทั้งจริงและเท็จ แสดงว่าชุดเบาะแสขัดแย้งกัน อย่าเลือกคำตอบโดยละเลยเบาะแส

13. ตัวอย่างโจทย์ครบขั้น: รวมจำนวนคู่

โจทย์: รับจำนวนเต็ม n ตามด้วยจำนวนเต็ม n ตัว แล้วแสดงผลรวมเฉพาะจำนวนคู่ เช่น input คือ 5 แล้วตามด้วย 1 2 3 4 5 ผลคือ 6 วิเคราะห์ก่อนเขียน: ต้องอ่าน n ครั้ง เก็บผลรวมเริ่มที่ศูนย์ ใช้เศษจากการหารสองตรวจความเป็นคู่ และพิมพ์เมื่ออ่านครบ หาก n=0 ผลคือ 0 ตัวอย่างนี้สมมติผลรวมไม่เกินขอบเขตชนิดข้อมูลที่เลือก

Python

import sys

data = list(map(int, sys.stdin.buffer.read().split()))
if data:
    n = data[0]
    values = data[1:1 + n]
    total = 0
    for value in values:
        if value % 2 == 0:
            total += value
    print(total)

sys.stdin.buffer.read() อ่านข้อมูลทั้งหมด split() แยกตามช่องว่างหรือบรรทัด map(int, ...) แปลงแต่ละรายการ และ list ทำให้เข้าถึงด้วย index/slice ได้ วิธีนี้ถือว่าโจทย์ให้ข้อมูลครบ n ค่า ใช้หน่วยความจำตามขนาด input หากข้อมูลใหญ่มากให้พิจารณาอ่านทีละบรรทัด

C

#include <stdio.h>
int main(void) {
    int n, value;
    long long total = 0;
    if (scanf("%d", &n) != 1) return 0;
    for (int i = 0; i < n; i++) {
        if (scanf("%d", &value) != 1) return 1;
        if (value % 2 == 0) total += value;
    }
    printf("%lld\n", total);
    return 0;
}

long long ให้ช่วงจำนวนเต็มมากกว่า int ในการใช้งานทั่วไป แต่ยังมีขีดจำกัด ตรวจข้อกำหนดโจทย์เสมอ ตัวแปร value ถูกอ่านใหม่แต่ละรอบ จึงไม่ต้องเก็บข้อมูลทุกตัว

Java

import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        if (!in.hasNextInt()) return;
        int n = in.nextInt();
        long total = 0;
        for (int i = 0; i < n; i++) {
            int value = in.nextInt();
            if (value % 2 == 0) total += value;
        }
        System.out.println(total);
    }
}

ทดสอบอย่างน้อยสามกรณี: 5 กับ 1 2 3 4 5 ได้ 6; 3 กับ −2 0 7 ได้ −2; 0 ได้ 0 หากจำนวนข้อมูลมาก Scanner อาจช้า จึงอาจใช้ BufferedReader และ StringTokenizer ตามข้อจำกัดของระบบสอบ

14. ตัวอย่างค้นหา: ตำแหน่งแรกของค่า

โจทย์ให้รายการและค่า target ต้องการตำแหน่งแรกโดยเริ่มนับจาก 0 หากไม่มีให้คืน −1 เริ่ม answer=−1 ไล่ i จาก 0 ถึง n−1 เมื่อพบให้กำหนด answer=i แล้วหยุด หากมีค่าซ้ำ การหยุดทันทีทำให้ได้ตำแหน่งแรก ต้นทุนกรณีแย่ O(n) และใช้พื้นที่เพิ่ม O(1) นอกเหนือจากรายการ

# Python
values = [4, 7, 4, 9]
target = 4
answer = -1
for i in range(len(values)):
    if values[i] == target:
        answer = i
        break
print(answer)  # 0
/* C: array a ความยาว n และ target กำหนดไว้แล้ว */
int answer = -1;
for (int i = 0; i < n; i++) {
    if (a[i] == target) { answer = i; break; }
}
// Java: array values และ target กำหนดไว้แล้ว
int answer = -1;
for (int i = 0; i < values.length; i++) {
    if (values[i] == target) { answer = i; break; }
}

หากโจทย์ต้องการนับตำแหน่งเริ่ม 1 ให้แปลงเฉพาะกรณีที่พบ อย่าเปลี่ยน −1 เป็น 0 ส่วน binary search ใช้เมื่อข้อมูลเรียงแล้ว และหากมีค่าซ้ำต้องปรับวิธีเพื่อหาตำแหน่งแรกโดยเฉพาะ

15. ข้อมูลแบบ map, set และการนับความถี่

เมื่ออยากทราบว่าแต่ละคำปรากฏกี่ครั้ง ใช้ map จากคำไปจำนวน เริ่มค่าที่ไม่เคยพบเป็น 0 แล้วเพิ่มหนึ่ง ใช้ตรวจคำซ้ำ โหวต หรือหาฐานนิยม หากต้องการเพียงรู้ว่าพบหรือไม่ใช้ set แทนได้

# Python
counts = {}
for word in ["cat", "dog", "cat"]:
    counts[word] = counts.get(word, 0) + 1
print(counts["cat"])  # 2
// Java: import java.util.HashMap;
HashMap<String, Integer> counts = new HashMap<>();
for (String word : new String[]{"cat", "dog", "cat"}) {
    counts.put(word, counts.getOrDefault(word, 0) + 1);
}
System.out.println(counts.get("cat"));

C ไม่มี map มาตรฐานแบบสำเร็จรูปในไลบรารีพื้นฐาน หากช่วงค่าจำกัด เช่น คะแนน 0–100 ใช้ int count[101] = {0}; แล้วเพิ่ม count[score] หลังตรวจช่วง หาก key เป็นข้อความหรือช่วงกว้าง ต้องเลือกโครงสร้างที่เหมาะกับข้อจำกัด

16. C pointers และหน่วยความจำที่ควรรู้

ตัวแปรมีค่าและตำแหน่งในหน่วยความจำ &x คือตำแหน่งของ x; int *p ประกาศ pointer ไปยัง int; *p อ่านหรือแก้ค่าที่ตำแหน่งซึ่ง p ชี้อยู่ เมื่อส่ง &x ให้ฟังก์ชัน ฟังก์ชันสามารถแก้ x ผ่าน pointer ได้ ต้องไม่ dereference pointer ที่ไม่ได้กำหนดหรือเป็น NULL

#include <stdio.h>
void add_one(int *p) {
    if (p != NULL) *p = *p + 1;
}
int main(void) {
    int x = 4;
    add_one(&x);
    printf("%d\n", x);  /* 5 */
    return 0;
}

ใน C การส่งอาร์เรย์ให้ฟังก์ชันมักส่งตัวชี้ไปสมาชิกแรก จึงต้องส่งความยาวแยก ห้ามอ่านเกินขอบเขต malloc จองหน่วยความจำแบบ dynamic และต้องตรวจผลก่อนใช้พร้อม free เมื่อเลิกใช้ โดยไม่ใช้ซ้ำหลัง free ถ้าโจทย์ขนาดเล็กและทราบขอบเขต การใช้อาร์เรย์ขนาดเหมาะสมมักเข้าใจง่ายกว่า
